A.3 最小可运行骨架

import math, torch, torch.nn as nn, torch.nn.functional as F

class Block(nn.Module):
    """一个解码器层，30 行。"""
    def __init__(self, d=64, h=4, d_ff=232, p=0.1):
        super().__init__()
        self.h, self.dk = h, d // h
        self.qkv = nn.Linear(d, 3 * d)        # 一次算完 QKV
        self.o   = nn.Linear(d, d)
        self.ff  = nn.Sequential(nn.Linear(d, d_ff), nn.ReLU(), nn.Linear(d_ff, d))
        self.n1, self.n2 = nn.LayerNorm(d), nn.LayerNorm(d)
        self.dp = nn.Dropout(p)

    def forward(self, x, mask):
        B, T, d = x.shape
        q, k, v = self.qkv(x).chunk(3, dim=-1)
        # 拆头
        q, k, v = [t.view(B, T, self.h, self.dk).transpose(1, 2) for t in (q, k, v)]
        # 式(1)
        s = q @ k.transpose(-2, -1) / math.sqrt(self.dk)
        s = s.masked_fill(~mask, float("-inf"))
        a = self.dp(F.softmax(s, dim=-1)) @ v
        # 拼头 + 残差 + LN
        a = a.transpose(1, 2).contiguous().view(B, T, d)
        x = self.n1(x + self.dp(self.o(a)))
        x = self.n2(x + self.dp(self.ff(x)))
        return x

A.6 训练不收敛的排查清单

按这个顺序查，不要跳：

  1. 数值：初始 loss 是否 ≈ ln(V)？学习率是否和论文公式对得上（用日志反算）？
  2. 数据：x 和 y 是否错位一位？有没有 token 全是 0？词表大小和模型配置一致吗？
  3. 梯度：打印几个参数的梯度范数。全为 0 说明断了，全是 nan 说明炸了。
  4. 结构：残差有没有加？LN 位置对不对？掩码方向反了吗（上三角 vs 下三角）？
  5. 容量：最后才怀疑这个。先确认前四项都对。
经验

80% 的「训练不收敛」问题出在第 1 步和第 2 步，只有不到 5% 出在第 5 步。 但初学者 90% 的时间花在第 5 步——加参数、加数据、加层数。顺序错了，努力全废。

A.7 一句话总结这本书

注意力是一次内容寻址的加权平均；
多头是在 h 个子空间里各做一次；
位置编码补上顺序；
残差和 LN 让深成为可能；
前馈在每个位置上做非线性加工；
交叉熵和梯度下降负责让它学会。
